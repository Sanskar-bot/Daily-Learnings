## Part 1 — End-to-End Integration Test: Full Access Flow

Wrote a test that exercises the complete patient→doctor workflow from registration
through to decrypted record access:

```python
# tests/test_e2e_access_flow.py

import pytest, requests, base64
from common.crypto_utils import generate_rsa_keypair, decrypt_wrapped_key, decrypt_record

BASE = "http://localhost:5000"

@pytest.fixture(scope="module")
def patient_session():
    """Register a patient, return their session token and key pair."""
    pub, priv = generate_rsa_keypair(key_size=4096)
    r = requests.post(f"{BASE}/auth/register", json={
        "name": "Test Patient", "email": "patient@test.com",
        "password": "TestPass123!", "role": "patient",
        "public_key": pub.decode()
    })
    assert r.status_code == 201
    token = r.json()["access_token"]
    return {"token": token, "pub": pub, "priv": priv}

@pytest.fixture(scope="module")
def doctor_session():
    """Register a doctor, return their session token and key pair."""
    pub, priv = generate_rsa_keypair(key_size=4096)
    r = requests.post(f"{BASE}/auth/register", json={
        "name": "Dr. Test", "email": "doctor@test.com",
        "password": "DoctorPass123!", "role": "doctor",
        "public_key": pub.decode()
    })
    assert r.status_code == 201
    token = r.json()["access_token"]
    return {"token": token, "pub": pub, "priv": priv}

def test_full_access_grant_flow(patient_session, doctor_session):
    pat, doc = patient_session, doctor_session
    pat_headers = {"Authorization": f"Bearer {pat['token']}"}
    doc_headers = {"Authorization": f"Bearer {doc['token']}"}

    # 1. Doctor requests access
    r = requests.post(f"{BASE}/api/access/request",
        json={"patient_id": pat["user_id"]}, headers=doc_headers)
    assert r.status_code == 201
    request_id = r.json()["request_id"]

    # 2. Patient approves — wraps AES key with doctor's public key
    aes_key = base64.b64decode(pat["aes_key_b64"])
    wrapped  = encrypt_key_for_doctor(aes_key, doc["pub"])
    r = requests.post(f"{BASE}/api/access/approve",
        json={"request_id": request_id, "wrapped_key": base64.b64encode(wrapped).decode()},
        headers=pat_headers)
    assert r.status_code == 200

    # 3. Doctor fetches patient data
    r = requests.get(f"{BASE}/emr/profile/{pat['user_id']}", headers=doc_headers)
    assert r.status_code == 200
    data = r.json()

    # 4. Doctor decrypts record locally
    wrapped_key = base64.b64decode(data["wrapped_key"])
    aes_key     = decrypt_wrapped_key(wrapped_key, doc["priv"])
    plaintext   = decrypt_record(base64.b64decode(data["ciphertext"]), aes_key)
    assert b"Test Patient" in plaintext
```

Running this test caught the identity mismatch bug (June 27) before it was fixed —
the `pat["user_id"]` in step 3 was being misread as `profile_code` by the backend.

## Part 2 — Persistent Rate Limiter

The in-memory rate limiter lost state on restart. A brute-force attacker could trigger
a server restart (e.g. crash via OOM) and immediately get a fresh window.

#### Fix — per-IP JSON files

```python
# server/server.py

_RATE_LIMITS_DIR = Path("server/rate_limits")
_RATE_LIMITS_DIR.mkdir(exist_ok=True)

def rate_limited(max_calls: int, window_seconds: int):
    def decorator(f):
        @wraps(f)
        def wrapper(*args, **kwargs):
            ip   = request.remote_addr or "unknown"
            path = _RATE_LIMITS_DIR / f"{ip}.json"

            now       = time.time()
            cutoff    = now - window_seconds
            # Load existing timestamps for this IP
            if path.exists():
                timestamps = [t for t in json.loads(path.read_text()) if t > cutoff]
            else:
                timestamps = []

            if len(timestamps) >= max_calls:
                retry_after = int(window_seconds - (now - timestamps[0]))
                return jsonify({"error": "rate_limited", "retry_after": retry_after}), 429

            timestamps.append(now)
            path.write_text(json.dumps(timestamps))
            return f(*args, **kwargs)
        return wrapper
    return decorator

# Usage
@app.route("/auth/login", methods=["POST"])
@rate_limited(max_calls=10, window_seconds=60)
def auth_login():
    ...
```

The hourly cleanup job deletes rate-limit files older than 10 minutes to prevent
disk accumulation:

```python
def _cleanup_rate_limits():
    cutoff = time.time() - 600   # 10 minutes
    for f in _RATE_LIMITS_DIR.glob("*.json"):
        if f.stat().st_mtime < cutoff:
            f.unlink(missing_ok=True)
```

## Part 3 — File Upload Validation (Magic Bytes)

The `/upload_image` endpoint previously accepted any file up to the Flask default
(no size limit, no type check). Added two layers of validation:

#### Layer 1 — Content-length cap

```python
app.config["MAX_CONTENT_LENGTH"] = 10 * 1024 * 1024   # 10 MB global cap

@app.errorhandler(413)
def request_too_large(e):
    return jsonify({"error": "file_too_large", "max_mb": 10}), 413
```

#### Layer 2 — Magic byte validation

File extensions can be spoofed (rename `malware.exe` to `photo.jpg`). Magic bytes
check the actual file content:

```python
MAGIC_BYTES = {
    b'\xff\xd8\xff':                    "image/jpeg",
    b'\x89PNG\r\n\x1a\n':              "image/png",
    b'GIF87a':                          "image/gif",
    b'GIF89a':                          "image/gif",
    b'RIFF':                            "image/webp",   # first 4 bytes; bytes 8-12 = WEBP
}

def _validate_image(file_storage, max_bytes=5 * 1024 * 1024) -> str:
    """Returns detected MIME type or raises ValueError."""
    header = file_storage.stream.read(12)
    file_storage.stream.seek(0)          # reset for actual save

    # Check size
    file_storage.stream.seek(0, 2)       # seek to end
    size = file_storage.stream.tell()
    file_storage.stream.seek(0)
    if size > max_bytes:
        raise ValueError(f"File too large: {size} bytes (max {max_bytes})")

    # Check magic bytes
    for magic, mime in MAGIC_BYTES.items():
        if header.startswith(magic):
            # Extra WebP check
            if mime == "image/webp" and header[8:12] != b'WEBP':
                continue
            return mime

    raise ValueError("Unsupported file type — must be JPEG, PNG, GIF, or WebP")
```

## Questions I Still Have

- The per-IP file rate limiter has a race condition under concurrent requests from the same IP (two requests read the same file simultaneously, both find 9/10 calls, both write 10/10 — effectively allowing 11). A file lock (`fcntl.flock`) would fix it on Linux. Windows needs a different approach.
- Should rate limits be per-IP, per-user, or both? A VPN or NAT gateway could share one IP across many legitimate users.
- Magic byte validation passes for a valid JPEG header followed by arbitrary data. A more thorough check would try to fully decode the image with Pillow — is that worth the latency?
